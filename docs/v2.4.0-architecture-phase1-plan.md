## 1. SessionState — corrected

### States (session-level)

```
IDLE       → initial state, no session active
STARTING   → session setup in progress (venv, model warmup)
RUNNING    → audio capture + ASR + overlay all active
PAUSED     → audio capture paused, overlay frozen
STOPPING   → graceful shutdown in progress
COMPLETED  → clean shutdown finished
ERROR      → unrecoverable error, session still exists for diagnostics
```

### Module-level status (in ModuleStatusRegistry, NOT SessionState)

```
UNINITIALIZED → module not yet initialized
STARTING      → initialization in progress
RUNNING       → normal operation
DEGRADED      → still working but with reduced capability (e.g. translation off)
STOPPING      → shutdown in progress
STOPPED       → cleanly stopped
ERROR         → failed, session may continue without this module
```

### SessionState class

```python
class SessionState:
    session_id: str           # uuid4
    state: SessionStateEnum
    started_at: float | None
    stopped_at: float | None
    error: str | None
    model_id: str             # "tiny"
    translation_mode: str     # "off" | "online" | "local"

    def can_transition(self, target: SessionStateEnum) -> bool
    def transition(self, target: SessionStateEnum) -> None  # raises on invalid
```

Invalid transitions raise `InvalidStateTransition`.

## 3. TranscriptEvent — full schema

```python
@dataclass
class TranscriptEvent:
    session_id: str            # uuid4, links to SessionState
    segment_id: str            # uuid4, stable across revisions
    utterance_id: int          # monotonic per session
    revision: int              # 1 = original, 2+ = edited/retranslated
    phase: Literal["PARTIAL", "STABLE", "FINAL"]
    seq: int                   # ordering within phase
    text_raw: str              # ASR raw output
    text_normalized: str | None  # post-processed (trimmed, deduped)
    text_user_edited: str | None # user-edit, None till edit happens
    translated_text: str | None
    source_lang: str | None
    target_lang: str | None
    start_time: float | None   # audio start offset in session
    end_time: float | None
    created_at: float          # epoch timestamp
    is_stale: bool             # True if newer revision supersedes this

    def with_edit(self, new_text: str) -> "TranscriptEvent"
    def with_translation(self, text: str) -> "TranscriptEvent"
    def mark_stale(self) -> "TranscriptEvent"
```

Phase 1 produces only PARTIAL (for real-time feedback) and FINAL (after silence).
STABLE is reserved for future use (e.g., short silence but utterance may continue).

## 4. TranslationScheduler — full design

### Guarantees (must be tested)

1. Only FINAL events are translated by default (PARTIAL pass-through)
2. Queue has max_size; overflow drops oldest
3. job_key = (session_id, segment_id, revision)
4. New revision cancels old pending job for same segment
5. Old-job results are discarded if revision is stale
6. Results from old session are discarded (session_id mismatch)
7. After Stop, no new jobs accepted
8. Translation failure does NOT stop ASR / overlay / storage

### Interface

```python
class TranslationScheduler:
    def __init__(self, config, module_registry, max_queue: int=20)

    def submit(self, event: TranscriptEvent) -> None
        # Phase 1: if translation_mode="off", returns immediately
        # Phase 1: if event.phase="PARTIAL", returns immediately (FINAL only)
        # Creates job_key, adds to queue, manages cancellation

    def cancel_all(self) -> None
    def pending_count(self) -> int

    # Internal
    def _worker_loop(self) -> None          # consumes queue
    def _is_stale(self, event: TranscriptEvent) -> bool  # compares revision + session + phase
    def _dispatch(self, event: TranscriptEvent) -> None  # placeholder: just logs in Phase 1
```

### Queue semantics

```
job = (session_id, segment_id, revision, event, submitted_at)

Insert:
  1. If queue is full → drop oldest
  2. If job with same (session_id, segment_id) exists and revision < current → cancel old
  3. Append new

Worker loop:
  1. Pop from queue
  2. Check _is_stale(job) → skip if True
  3. _dispatch(job) → Phase 1: log "[TranslationScheduler] would translate {segment_id}"
  4. Notify callback (future: write result into event + store in DB)
```

## 5. SessionRepository — SQLite schema

### Tables

```sql
CREATE TABLE IF NOT EXISTS sessions (
    session_id TEXT PRIMARY KEY,
    state TEXT NOT NULL DEFAULT 'IDLE',
    model_id TEXT NOT NULL,
    translation_mode TEXT NOT NULL DEFAULT 'off',
    source_lang TEXT,
    target_lang TEXT,
    started_at REAL,
    stopped_at REAL,
    error TEXT,
    created_at REAL NOT NULL,
    updated_at REAL NOT NULL
);

CREATE TABLE IF NOT EXISTS segments (
    segment_id TEXT PRIMARY KEY,
    session_id TEXT NOT NULL REFERENCES sessions(session_id),
    utterance_id INTEGER NOT NULL,
    phase TEXT NOT NULL,            -- PARTIAL, STABLE, FINAL
    seq INTEGER NOT NULL,
    text_raw TEXT NOT NULL,
    text_normalized TEXT,
    text_user_edited TEXT,
    source_lang TEXT,
    start_time REAL,
    end_time REAL,
    revision INTEGER NOT NULL DEFAULT 1,
    is_stale INTEGER NOT NULL DEFAULT 0,
    created_at REAL NOT NULL,
    updated_at REAL NOT NULL
);

CREATE TABLE IF NOT EXISTS segment_revisions (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    segment_id TEXT NOT NULL REFERENCES segments(segment_id),
    revision INTEGER NOT NULL,
    text_raw TEXT NOT NULL,
    text_user_edited TEXT,
    action TEXT NOT NULL,           -- 'asr', 'edit', 'retranslate'
    created_at REAL NOT NULL
);

CREATE TABLE IF NOT EXISTS translations (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    segment_id TEXT NOT NULL REFERENCES segments(segment_id),
    session_id TEXT NOT NULL REFERENCES sessions(session_id),
    revision INTEGER NOT NULL,
    source_text TEXT NOT NULL,
    translated_text TEXT,
    status TEXT NOT NULL DEFAULT 'NOT_REQUESTED',
    -- status: NOT_REQUESTED, QUEUED, RUNNING, COMPLETED, FAILED, STALE, CANCELLED
    error TEXT,
    started_at REAL,
    completed_at REAL,
    created_at REAL NOT NULL
);

CREATE TABLE IF NOT EXISTS module_events (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    session_id TEXT NOT NULL REFERENCES sessions(session_id),
    module_name TEXT NOT NULL,
    event_type TEXT NOT NULL,       -- 'status_change', 'error', 'degraded', 'recovery'
    old_status TEXT,
    new_status TEXT,
    detail TEXT,
    created_at REAL NOT NULL
);

CREATE INDEX IF NOT EXISTS idx_segments_session ON segments(session_id, seq);
CREATE INDEX IF NOT EXISTS idx_segments_utterance ON segments(session_id, utterance_id);
CREATE INDEX IF NOT EXISTS idx_translations_segment ON translations(segment_id, revision);
CREATE INDEX IF NOT EXISTS idx_translations_status ON translations(status);
CREATE INDEX IF NOT EXISTS idx_module_events_session ON module_events(session_id);
```

### SessionRepository interface

```python
class SessionRepository:
    def __init__(self, db_path: str)

    # Sessions
    def create_session(self, session: SessionState) -> None
    def update_session(self, session: SessionState) -> None
    def get_session(self, session_id: str) -> SessionState | None
    def list_sessions(self, limit: int=20) -> List[SessionState]

    # Segments
    def save_segment(self, event: TranscriptEvent) -> None
    def get_segments(self, session_id: str) -> List[TranscriptEvent]
    def get_segment(self, segment_id: str) -> TranscriptEvent | None
    def get_segment_history(self, segment_id: str) -> List[TranscriptEvent]

    # Translations
    def save_translation_status(self, segment_id: str, session_id: str, revision: int, status: str) -> None
    def save_translation_result(self, segment_id: str, session_id: str, revision: int, translated_text: str) -> None

    # Module events
    def log_module_event(self, session_id: str, module_name: str, event_type: str,
                         old_status: str|None, new_status: str|None, detail: str|None) -> None
```
